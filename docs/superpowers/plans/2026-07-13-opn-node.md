# n8n-nodes-opn Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `n8n-nodes-opn` — Opn (Omise) payments community node package: credential + declarative action node (7 ops) + webhook trigger with verify-by-refetch — published to npm, submitted to the n8n registry, demoed on n8n.phangan.ai before 2026-07-22.

**Architecture:** Standard n8n community-node package (n8n-nodes-starter layout). Action node = declarative style (routing objects, no execute method). Trigger = programmatic webhook node. Zero runtime dependencies; all HTTP through n8n helpers/routing.

**Tech Stack:** TypeScript, n8n-workflow (peer), jest + ts-jest, eslint-plugin-n8n-nodes-base, prettier, gulp (icon copy), GitHub Actions.

## Global Constraints

- npm name `n8n-nodes-opn`; fallback `n8n-nodes-opn-payments` (Task 1 verifies).
- **Zero runtime dependencies** (n8n verified-node rule). devDependencies only.
- Keyword `n8n-community-node-package` MUST be in package.json (registry requirement).
- Base URL `https://api.omise.co`; auth = HTTP Basic, username = secret key, empty password.
- Amounts are subunits (satang/cents) — documented in field descriptions, never converted.
- Demo and tests use TEST keys (`skey_test_…`) only; no key ever committed or logged.
- Node.js >= 18.10. License MIT. Author Guillaume Flambard.
- Every commit: conventional message, code formatted (`npm run format`), lint clean.

```ts
import { Opn } from '../nodes/Opn/Opn.node';
import type { INodeProperties } from 'n8n-workflow';

const node = new Opn();
const props = node.description.properties;

function operationOptions(resource: string): INodeProperties {
	const op = props.find(
		(p) =>
			p.name === 'operation' &&
			(p.displayOptions?.show?.resource as string[])?.includes(resource),
	);
	if (!op) throw new Error(`no operation selector for ${resource}`);
	return op;
}

function routingOf(resource: string, operation: string) {
	const op = operationOptions(resource);
	const choice = (op.options as Array<{ value: string; routing?: unknown }>).find(
		(o) => o.value === operation,
	);
	if (!choice) throw new Error(`${resource}.${operation} not found`);
	return (choice as { routing: { request: { method: string; url: string } } }).routing.request;
}

describe('Opn node description', () => {
	test('declarative base', () => {
		expect(node.description.name).toBe('opn');
		expect(node.description.requestDefaults?.baseURL).toBe('https://api.omise.co');
		expect(node.description.credentials).toEqual([{ name: 'opnApi', required: true }]);
	});

	test('charge operations route to Omise endpoints', () => {
		expect(routingOf('charge', 'create')).toEqual({ method: 'POST', url: '/charges' });
		expect(routingOf('charge', 'get')).toEqual({
			method: 'GET',
			url: '=/charges/{{$parameter.chargeId}}',
		});
		expect(routingOf('charge', 'getAll')).toEqual({ method: 'GET', url: '/charges' });
		expect(routingOf('charge', 'capture')).toEqual({
			method: 'POST',
			url: '=/charges/{{$parameter.chargeId}}/capture',
		});
	});

	test('charge create sends amount and currency in the body', () => {
		const amount = props.find((p) => p.name === 'amount' && p.displayOptions?.show?.resource?.includes('charge'));
		expect(amount?.routing?.send).toEqual({ type: 'body', property: 'amount' });
		expect(amount?.description).toContain('subunits');
		const currency = props.find((p) => p.name === 'currency');
		expect(currency?.routing?.send).toEqual({ type: 'body', property: 'currency' });
	});
});
```

```ts
import type {
	IDataObject,
	INodeType,
	INodeTypeDescription,
	IWebhookFunctions,
	IWebhookResponseData,
} from 'n8n-workflow';

const EVENT_OPTIONS = [
	{ name: 'All Events', value: '*' },
	{ name: 'Charge Complete', value: 'charge.complete' },
	{ name: 'Charge Create', value: 'charge.create' },
	{ name: 'Charge Update', value: 'charge.update' },
	{ name: 'Customer Create', value: 'customer.create' },
	{ name: 'Refund Create', value: 'refund.create' },
	{ name: 'Transfer Pay', value: 'transfer.pay' },
];

export class OpnTrigger implements INodeType {
	description: INodeTypeDescription = {
		displayName: 'Opn Trigger',
		name: 'opnTrigger',
		icon: 'file:opn.svg',
		group: ['trigger'],
		version: 1,
		description: 'Starts a workflow on Opn (Omise) webhook events',
		defaults: { name: 'Opn Trigger' },
		inputs: [],
		outputs: ['main'],
		credentials: [{ name: 'opnApi', required: true }],
		webhooks: [{ name: 'default', httpMethod: 'POST', responseMode: 'onReceived', path: 'webhook' }],
		properties: [
			{
				displayName:
					'Register this webhook URL in your Opn dashboard (Settings → Webhooks). Opn sends every event to it; this node filters and verifies.',
				name: 'setupNotice',
				type: 'notice',
				default: '',
			},
			{
				displayName: 'Events',
				name: 'events',
				type: 'multiOptions',
				default: ['*'],
				options: EVENT_OPTIONS,
				description: 'Only emit for these event keys',
			},
			{
				displayName: 'Verify Event',
				name: 'verifyEvent',
				type: 'boolean',
				default: true,
				description:
					'Whether to re-fetch the event from the Opn API before emitting. Opn webhooks are unsigned; verification prevents spoofed payloads.',
			},
		],
	};

	webhookMethods = {
		default: {
			// Webhook endpoint is registered manually in the Opn dashboard — nothing to create or
			// delete on Opn's side, so activation just accepts the static URL.
			async checkExists(): Promise<boolean> {
				return true;
			},
			async create(): Promise<boolean> {
				return true;
			},
			async delete(): Promise<boolean> {
				return true;
			},
		},
	};

	async webhook(this: IWebhookFunctions): Promise<IWebhookResponseData> {
		const body = this.getBodyData() as IDataObject;
		if (body?.object !== 'event' || typeof body.id !== 'string') {
			return {};
		}

		const events = this.getNodeParameter('events', ['*']) as string[];
		const key = body.key as string;
		if (!events.includes('*') && !events.includes(key)) {
			return {};
		}

		let payload: IDataObject = body;
		const verifyEvent = this.getNodeParameter('verifyEvent', true) as boolean;
		if (verifyEvent) {
			try {
				payload = (await this.helpers.httpRequestWithAuthentication.call(this, 'opnApi', {
					method: 'GET',
					url: `https://api.omise.co/events/${body.id}`,
					json: true,
				})) as IDataObject;
			} catch {
				// Unverifiable event (spoofed, deleted, or wrong account) — acknowledge and drop.
				return {};
			}
		}

		return {
			workflowData: [this.helpers.returnJsonArray(payload)],
		};
	}
}
```
